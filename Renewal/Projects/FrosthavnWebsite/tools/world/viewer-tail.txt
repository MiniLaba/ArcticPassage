  if(now-drawAt<33){requestDraw();return;}
  const dt=last?Math.min(.1,(now-last)/1000):1/30;last=now;drawAt=now;
  if(!options.reducedMotion)simulation+=dt;
  let sunlight=manualSun;
  if(cycle&&!options.reducedMotion){
   year=(year+dt*.47)%30;const s=yearState(year);goal={snow:s.snow,autumn:s.autumn,night:s.night};sunlight=s.sunDirection;
   const next=year<3.8?'winter':year<11?'spring':year<18.1?'summer':year<24.4?'autumn':'winter';
   if(next!==activeSeason){activeSeason=next;options.onSeason?.(next);}
  }
  for(const key of ['snow','night','autumn'])current[key]=mix(current[key],goal[key],options.reducedMotion?1:1-Math.exp(-dt*3));
  lighting(simulation,current.snow,current.night,current.autumn,.22,manualSun?.58:.2+.35*current.autumn,sunlight);
  if(transition){
   const q=smooth(0,options.reducedMotion?1:1250,now-transition.start);camera.position.copy(transition.from).lerp(transition.to,q);controls.target.copy(transition.fromTarget).lerp(transition.toTarget,q);
   if(q>=1)transition=null;
  }
